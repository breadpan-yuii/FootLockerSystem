(D) COPY-PASTE SOURCE FOR dbdiagram.io (DBML)

Paste the block below at https://dbdiagram.io and it renders the ERD:

Table users {
  id bigint [pk]
  name varchar
  email varchar [unique]
  email_verified_at timestamp [null]
  password varchar
  role varchar [note: 'admin | customer']
  phone varchar [null]
  address text [null]
  remember_token varchar [null]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table brands {
  id bigint [pk]
  name varchar
  slug varchar [unique]
  logo varchar [null]
  description text [null]
  is_active boolean [default: true]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table categories {
  id bigint [pk]
  name varchar
  slug varchar [unique]
  description text [null]
  image varchar [null]
  gender varchar [note: 'men | women | kids | unisex']
  is_active boolean [default: true]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table products {
  id bigint [pk]
  brand_id bigint
  category_id bigint
  name varchar
  slug varchar [unique]
  description text [null]
  price decimal(10,2)
  sale_price decimal(10,2) [null]
  stock int [default: 0]
  image varchar [null]
  gender varchar [note: 'men | women | kids | unisex']
  is_featured boolean [default: false]
  is_active boolean [default: true]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table product_images {
  id bigint [pk]
  product_id bigint
  image varchar
  is_primary boolean [default: false]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table product_variants {
  id bigint [pk]
  product_id bigint
  size varchar
  color varchar [null]
  sku varchar [unique, null]
  stock int [default: 0]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table carts {
  id bigint [pk]
  user_id bigint [unique]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table cart_items {
  id bigint [pk]
  cart_id bigint
  product_id bigint
  product_variant_id bigint [null]
  quantity int [default: 1]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table orders {
  id bigint [pk]
  user_id bigint
  order_number varchar [unique]
  subtotal decimal(10,2) [default: 0]
  shipping_fee decimal(10,2) [default: 0]
  total_amount decimal(10,2)
  status varchar [note: 'pending|confirmed|preparing|completed|cancelled']
  order_date datetime
  full_name varchar
  phone varchar
  address text
  payment_method varchar [default: 'Cash on Delivery']
  notes text [null]
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table order_items {
  id bigint [pk]
  order_id bigint
  product_id bigint
  product_variant_id bigint [null]
  quantity int [default: 1]
  price decimal(10,2)
  subtotal decimal(10,2)
  created_at timestamp [null]
  updated_at timestamp [null]
}

Table wishlist {
  id bigint [pk]
  user_id bigint
  product_id bigint
  created_at timestamp [null]
  updated_at timestamp [null]
}

// ---- Relationships ----
Ref: products.brand_id > brands.id
Ref: products.category_id > categories.id
Ref: product_images.product_id > products.id
Ref: product_variants.product_id > products.id
Ref: carts.user_id - users.id
Ref: cart_items.cart_id > carts.id
Ref: cart_items.product_id > products.id
Ref: cart_items.product_variant_id > product_variants.id
Ref: orders.user_id > users.id
Ref: order_items.order_id > orders.id
Ref: order_items.product_id > products.id
Ref: order_items.product_variant_id > product_variants.id
Ref: wishlist.user_id > users.id
Ref: wishlist.product_id > products.id
